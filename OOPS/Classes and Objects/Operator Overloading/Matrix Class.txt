#include<iostream>
using namespace std;
class Matrix
{
private:
    int rows,cols;
    int **a;
public:
    Matrix(int r,int c):rows(r),cols(c)
    {
        a=new int*[rows];
        for(int i=0;i<rows;i++)
            a[i]=new int[cols]();
    }
    ~Matrix()
    {
        for(int i=0;i<rows;i++)
            delete[] a[i];
        delete[] a;
    }
    int * operator[](int index)
    {
        if(index>rows || index<0)
        {
            cout<<"Error:Out Of Bounds"<<endl;
            return NULL;
        }
        return a[index];
    }
    const int * operator[](int index)const            //note the return type
    {
        if(index>rows || index<0)
        {
            cout<<"Error:Out Of Bounds"<<endl;
            return NULL;
        }
        return a[index];
    }
    void display()const
    {
        for(int i=0;i<rows;i++)
        {
            for(int j=0;j<cols;j++)
                cout<<a[i][j]<<" ";
            cout<<endl;
        }
    }
};

int main()
{
    Matrix m(3,3);
    m[0][0] = 1;
    m[0][1] = 2;
    m[0][2] = 3;
    m[1][0] = 4;
    m[1][1] = 5;
    m[1][2] = 6;
    m[2][0] = 7;
    m[2][1] = 8;
    m[2][2] = 9;
    m.display();
}
