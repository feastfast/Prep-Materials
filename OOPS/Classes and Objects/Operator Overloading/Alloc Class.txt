Example 1 : Without Arrays
#include<iostream>
#include<cstdlib>
using namespace std;
class Alloc
{
public:
    const int *data;
    Alloc(int x=0):data(new int(x)){}
    ~Alloc()
    {
        delete data;
    }
    void* operator new(size_t s)
    {
        //allocating s bytes
        void *ptr=malloc(s);
        if(!ptr)
        {
            cout<<"Allocation Failed"<<endl;
            return NULL;
        }
        return ptr;
    }

    void operator delete(void *ptr)
    {
        free(ptr);
    }

    void* operator new[](size_t s)
    {
        void *ptr=malloc(s);
        if(!ptr)
        {
            cout<<"Allocation Failed"<<endl;
            return NULL;
        }
        return ptr;
    }

    void operator delete[](void *ptr)
    {
        free(ptr);
    }

};

int main()
{
    Alloc *o=new Alloc[3]{Alloc(3),Alloc(4),Alloc(5)};
    cout<<*(o[0].data)<<endl<<*(o[1].data)<<endl<<(*o[2].data)<<endl;
    delete[] o;
}

Example 2 : With Arrays

#include<iostream>
#include<cstdlib>
using namespace std;
class Alloc
{
public:
    int *data;
    Alloc(int x=0):data(new int[5]())
    {
        for(int i=0;i<5;i++)
            data[i]=x;
    }
    ~Alloc()
    {
        delete[] data;
    }
    void* operator new(size_t s)
    {
        //allocating s bytes
        void *ptr=malloc(s);
        if(!ptr)
        {
            cout<<"Allocation Failed"<<endl;
            return NULL;
        }
        return ptr;
    }

    void operator delete(void *ptr)
    {
        free(ptr);
    }

    void* operator new[](size_t s)
    {
        void *ptr=malloc(s);
        if(!ptr)
        {
            cout<<"Allocation Failed"<<endl;
            return NULL;
        }
        return ptr;
    }

    void operator delete[](void *ptr)
    {
        free(ptr);
    }

    void print()
    {
        for(int i=0;i<5;i++)
            cout<<data[i]<<" ";
        cout<<endl;
    }

};

int main()
{
    Alloc *o=new Alloc[3]{Alloc(3),Alloc(4),Alloc(5)};
    o[0].print();
    o[1].print();
    o[2].print();
    delete[] o;
}
